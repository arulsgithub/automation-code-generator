// Imports needed (if missing): org.openqa.selenium.WebElement, org.openqa.selenium.support.FindBy, org.openqa.selenium.By

    @FindBy(css = "[data-test='cancel']")
    private WebElement cancelButton;

    public LoginPage removeItem(String itemName) {
        String id = "remove-" + itemName.toLowerCase().replace(' ', '-');
        click(driver.findElement(By.cssSelector("[data-test='" + id + "']")));
        return this;
    }

    public void clickCancel() {
        click(cancelButton);
    }
